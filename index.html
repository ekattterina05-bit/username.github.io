    <script>
        let score = 0;
        let selectedOptions = {
            step1: null,
            step2: null,
            step3: null
        };

        // Функция для сброса состояния кнопок
        function resetNextButton(stepId) {
            document.getElementById(`next${stepId}`).disabled = true;
        }

        // Функция для переключения шагов
        function goToStep(currentStepId, nextStepId) {
            document.getElementById(`step${currentStepId}`).classList.remove('active');
            document.getElementById(`step${nextStepId}`).classList.add('active');
        }

        // Шаг 1: Выбор аптечки
        document.querySelectorAll('#step1 .option').forEach(option => {
            option.addEventListener('click', () => {
                document.querySelectorAll('#step1 .option').forEach(o => o.classList.remove('selected'));
                option.classList.add('selected');
                selectedOptions.step1 = option.dataset.option;
                document.getElementById('next1').disabled = false;
            });
        });

        document.getElementById('next1').addEventListener('click', () => {
            const feedback = document.getElementById('feedback1');
            if (selectedOptions.step1 === 'aptechka') {
                feedback.textContent = 'Правильно! Аптечка — обязательный атрибут.';
                feedback.className = 'feedback success';
                score += 10;
                document.getElementById('score').textContent = score;
            } else {
                feedback.textContent = 'Неверно! Аптечка должна быть всегда под рукой.';
                feedback.className = 'feedback error';
            }
            goToStep(1, 2);
            resetNextButton(2);
        });

        // Шаг 2: Выбор домкрата
        document.querySelectorAll('#step2 .option').forEach(option => {
            option.addEventListener('click', () => {
                document.querySelectorAll('#step2 .option').forEach(o => o.classList.remove('selected'));
                option.classList.add('selected');
                selectedOptions.step2 = option.dataset.option;
                document.getElementById('next2').disabled = false;
            });
        });

        document.getElementById('next2').addEventListener('click', () => {
            const feedback = document.getElementById('feedback2');
            if (selectedOptions.step2 === 'domkrat') {
                feedback.textContent = 'Верно! Домкрат выручит при проколе колеса.';
                feedback.className = 'feedback success';
                score += 10;
                document.getElementById('score').textContent = score;
            } else {
                feedback.textContent = 'Ошибка! Домкрат необходим для замены колеса.';
                feedback.className = 'feedback error';
            }
            goToStep(2, 3);
            resetNextButton(3);
        });

        // Шаг 3: Выбор свечей зажигания
        document.querySelectorAll('#step3 .option').forEach(option => {
            option.addEventListener('click', () => {
                document.querySelectorAll('#step3 .option').forEach(o => o.classList.remove('selected'));
                option.classList.add('selected');
                selectedOptions.step3 = option.dataset.option;
                document.getElementById('finish').disabled = false;
            });
        });

        document.getElementById('finish').addEventListener('click', () => {
            const feedback = document.getElementById('feedback3');
            if (selectedOptions.step3 === 'svechi') {
                feedback.textContent = 'Отлично! Свечи зажигания пригодятся в дороге.';
                feedback.className = 'feedback success';
                score += 10;
                document.getElementById('score').textContent = score;
            } else {
                feedback.textContent = 'Неправильно! Свечи зажигания могут спасти двигатель.';
                feedback.className = 'feedback error';
            }
            document.getElementById('step3').classList.remove('active');
            document.getElementById('final').classList.add('active');
            document.getElementById('final-score').textContent = score;

            // Текст результата
            const resultText = document.getElementById('result-text');
            if (score === 30) {
                resultText.textContent = 'Ты эксперт! Собрал идеальный набор автомобилиста.';
            } else if (score >= 20) {
                resultText.textContent = 'Хорошо! Но можно лучше — проверь набор ещё раз.';
            } else {
                resultText.textContent = 'Пока не очень... Попробуй ещё раз!';
            }
        });

        // Рестарт игры
        document.getElementById('restart').addEventListener('click', () => {
            score = 0;
            document.getElementById('score').textContent = score;
            document.querySelectorAll('.option').forEach(option => {
                option.classList.remove('selected');
            });
            document.querySelectorAll('.feedback').forEach(feedback => {
                feedback.className = 'feedback';
                feedback.textContent = '';
            });
            document.getElementById('final').classList.remove('active');
            document.getElementById('step1').classList.add('active');
            document.getElementById('next1').disabled = true;
            document.getElementById('next2').disabled = true;
            document.getElementById('finish').disabled = true;
        });
